SET SERVEROUTPUT ON;

DECLARE
    CURSOR order_cursor IS
        SELECT Order_ID, Equipment, Variant, Amount
        FROM Orders;
BEGIN
    FOR order_rec IN order_cursor LOOP

        IF order_rec.Amount > 10000 THEN
            DBMS_OUTPUT.PUT_LINE(
                'Order ' || order_rec.Order_ID ||
                ' - ' || order_rec.Equipment ||
                ' (' || order_rec.Variant || ')' ||
                ': High Value'
            );
        END IF;

    END LOOP;
END;
