select * from GEN

SET SERVEROUTPUT ON;

DECLARE
    CURSOR gen_cursor IS
        SELECT gen_name, gen_skill
        FROM gen
        ORDER BY gen_skill DESC;

    v_gen_name gen.gen_name%TYPE;
    v_salary   gen.gen_skill%TYPE;

BEGIN
    OPEN gen_cursor;

    FOR i IN 1..5 LOOP
        FETCH gen_cursor INTO v_gen_name, v_salary;

        EXIT WHEN gen_cursor%NOTFOUND;

        DBMS_OUTPUT.PUT_LINE(
            'Name: ' || v_gen_name ||
            ' | Salary: ' || v_salary
        );
    END LOOP;

    CLOSE gen_cursor;
END;
